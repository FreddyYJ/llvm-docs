28/// Base class for all values computed by abstract interpretation.
29class Value {
30public:
31 enum class Kind { Bool, Integer, Reference, Pointer, Struct };
32
33 explicit Value(Kind ValKind) : ValKind(ValKind) {}
34
35 virtual ~Value() = default;
36
37 Kind getKind() const { return ValKind; }
38
39private:
40 Kind ValKind;
41};

43/// Models a boolean.
44class BoolValue : public Value {
45public:
46 explicit BoolValue() : Value(Kind::Bool) {}
47
48 static bool classof(const Value *Val) { return Val->getKind() == Kind::Bool; }
49};

51/// Models an integer.
52class IntegerValue : public Value {
53public:
54 explicit IntegerValue() : Value(Kind::Integer) {}
55
56 static bool classof(const Value *Val) {
57 return Val->getKind() == Kind::Integer;
58 }
59};

61/// Base class for values that refer to storage locations.
62class IndirectionValue : public Value {
63public:
64 /// Constructs a value that refers to `PointeeLoc`.
65 explicit IndirectionValue(Kind ValueKind, StorageLocation &PointeeLoc)
66 : Value(ValueKind), PointeeLoc(PointeeLoc) {}
67
68 static bool classof(const Value *Val) {
69 return Val->getKind() == Kind::Reference || Val->getKind() == Kind::Pointer;
70 }
71
72 StorageLocation &getPointeeLoc() const { return PointeeLoc; }
73
74private:
75 StorageLocation &PointeeLoc;
76};

78/// Models a dereferenced pointer. For example, a reference in C++ or an lvalue
79/// in C.

101/// Models a value of `struct` or `class` type.
102class StructValue final : public Value {
103public:
104 StructValue() : StructValue(llvm::DenseMap<const ValueDecl *, Value *>()) {}
105
106 explicit StructValue(llvm::DenseMap<const ValueDecl *, Value *> Children)
107 : Value(Kind::Struct), Children(std::move(Children)) {}
108
109 static bool classof(const Value *Val) {
110 return Val->getKind() == Kind::Struct;
111 }
112
113 /// Returns the child value that is assigned for `D`.
114 Value &getChild(const ValueDecl &D) const {
115 auto It = Children.find(&D);
116 assert(It != Children.end());
117 return *It->second;
118 }
119
120 /// Assigns `Val` as the child value for `D`.
121 void setChild(const ValueDecl &D, Value &Val) { Children[&D] = &Val; }
122
123 /// Returns the value of the synthetic property with the given `Name` or null
124 /// if the property isn't assigned a value.
125 Value *getProperty(llvm::StringRef Name) const {
126 auto It = Properties.find(Name);
127 return It == Properties.end() ? nullptr : It->second;
128 }
129
130 /// Assigns `Val` as the value of the synthetic property with the given
131 /// `Name`.
132 void setProperty(llvm::StringRef Name, Value &Val) {
133 Properties.insert_or_assign(Name, &Val);
134 }
135
136private:
137 llvm::DenseMap<const ValueDecl *, Value *> Children;
138 llvm::StringMap<Value *> Properties;
139};
